```bash
import re, os, json

session_dir = "<session_dir>"
ip_output = f"{session_dir}/geo/collected_ips.json"

# Private IP ranges to EXCLUDE
PRIVATE_PREFIXES = [
    ("10.", "RFC1918"),
    ("172.16.", "RFC1918"),
    ("172.17.", "RFC1918"),
    ("172.18.", "RFC1918"),
    ("172.19.", "RFC1918"),
    ("172.20.", "RFC1918"),
    ("172.21.", "RFC1918"),
    ("172.22.", "RFC1918"),
    ("172.23.", "RFC1918"),
    ("172.24.", "RFC1918"),
    ("172.25.", "RFC1918"),
    ("172.26.", "RFC1918"),
    ("172.27.", "RFC1918"),
    ("172.28.", "RFC1918"),
    ("172.29.", "RFC1918"),
    ("172.30.", "RFC1918"),
    ("172.31.", "RFC1918"),
    ("192.168.", "RFC1918"),
    ("127.", "loopback"),
    ("0.", "reserved"),
]

def is_private(ip):
    for prefix, reason in PRIVATE_PREFIXES:
        if ip.startswith(prefix):
            return True, reason
    return False, None

IP_PATTERN = r'\b(?:(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\.){3}(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\b'

seen = set()

# Scan email headers for IPs
email_dirs = [f"{session_dir}/crawl/raw", f"{session_dir}/discovery"]
for edir in email_dirs:
    if not os.path.exists(edir): continue
    for root, dirs, files in os.walk(edir):
        for fname in files:
            fpath = os.path.join(root, fname)
            try:
                with open(fpath, 'r', errors='replace') as f:
                    content = f.read()
                
                # Extract IPs from Received headers
                for line in content.split('\n'):
                    if line.strip().startswith('Received:'):
                        ips = re.findall(IP_PATTERN, line)
                        for ip in ips:
                            priv, reason = is_private(ip)
                            if not priv and ip not in seen:
                                seen.add(ip)
                                collected.append({
                                    "ip": ip,
                                    "source": "email_header",
                                    "source_file": fpath,
                                    "header_line": line.strip()[:200]
                                })
            except: pass

# Scan infrastructure results for IPs
infra_file = f"{session_dir}/infra/infra_results.json"
if os.path.exists(infra_file):
    try:
        with open(infra_file) as f:
            infra = json.load(f)
        for ip_entry in infra.get("ip_resolution", []):
            ip = ip_entry.get("ip", "")
            if ip and ip not in seen:
                priv, _ = is_private(ip)
                if not priv:
                    seen.add(ip)
                    collected.append({
                        "ip": ip,
                        "source": "infra_resolution",
                        "domain": ip_entry.get("domain", "")
                    })
    except: pass

with open(ip_output, "w") as f:
    json.dump(collected, f, indent=2)

print(f"Collected {len(collected)} unique public IPs for geolocation")
for c in collected:
    print(f"  IP: {c['ip']} (source: {c['source']})")
```

#### 4a. SunCalc Computation Script
```bash
# SunCalc shadow analysis — actual Python implementation
# This requires: pip install suncalc  (or use the math below)

python3 << 'PYEOF'
import json, os, sys
from datetime import datetime, timezone
import math

session_dir = "<session_dir>/geo"

with open(f"{session_dir}/exif_analysis.json") as f:
    exif_results = json.load(f)

# Find images WITHOUT GPS but WITH a timestamp
no_gps_with_time = [
    r for r in exif_results 
    if r.get("gps_status") == "NO_GPS_DATA" and r.get("datetime_original")
]

if not no_gps_with_time:
    print("No images without GPS but with timestamp found for SunCalc analysis")
    sys.exit(0)

print(f"Found {len(no_gps_with_time)} images for SunCalc shadow analysis")

# Sun position calculation (no external library required)
def sun_position(lat, lon, dt):
    """Calculate solar azimuth and elevation for given lat/lon/time."""
    # Day of year
    doy = dt.timetuple().tm_yday
    
    # Solar declination (Cooper's equation)
    declination = 23.45 * math.sin(math.radians(360 / 365 * (doy - 81)))
    
    # Hour angle
    solar_noon = 12.0 - lon / 15.0  # in UTC hours
    hour_angle = (dt.hour + dt.minute / 60.0 + dt.second / 3600.0 - solar_noon) * 15.0
    
    # Convert to radians
    lat_r = math.radians(lat)
    dec_r = math.radians(declination)
    ha_r = math.radians(hour_angle)
    
    # Solar elevation
    sin_elev = (math.sin(lat_r) * math.sin(dec_r) + 
                math.cos(lat_r) * math.cos(dec_r) * math.cos(ha_r))
    elevation = math.degrees(math.asin(max(-1, min(1, sin_elev))))
    
    # Solar azimuth
    cos_az = (math.sin(dec_r) - math.sin(lat_r) * sin_elev) / \
             (math.cos(lat_r) * math.cos(math.radians(elevation)) + 1e-10)
    azimuth = math.degrees(math.acos(max(-1, min(1, cos_az))))
    
    if hour_angle > 0:
        azimuth = 360 - azimuth
    
    return {"elevation": elevation, "azimuth": azimuth, "declination": declination}

# Shadow direction from azimuth
# Shadow falls OPPOSITE to the sun's azimuth
# So shadow_direction = (azimuth + 180) % 360
def shadow_direction(azimuth):
    return (azimuth + 180) % 360

def direction_name(degrees):
    dirs = ["N", "NNE", "NE", "ENE", "E", "ESE", "SE", "SSE",
```
