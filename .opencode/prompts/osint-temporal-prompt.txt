## Timestamp Normalization Script

Save this as `/tmp/osint_temporal_normalize.py`. This is a COMPLETE, WORKING Python script that handles 15+ timestamp formats and validates against future dates.

```python
#!/usr/bin/env python3
"""
OSINT Temporal — Timestamp normalization across 15+ formats.
Validates against future dates. Normalizes to UTC ISO 8601.
Usage: python3 /tmp/osint_temporal_normalize.py <input_events.json> <output_normalized.json>
Input: JSON file with array of events, each having a "timestamp" field (any format)
Output: JSON file with normalized timestamps and validation results
"""
import json
import sys
import os
import re
import math
from datetime import datetime, timezone, timedelta
from typing import Optional, List, Dict, Any

# Current UTC time — used for future-date validation
NOW_UTC = datetime.now(timezone.utc)

# Supported format parsers — each returns a datetime object or None

def parse_iso8601(ts: str) -> Optional[datetime]:
    """ISO 8601: 2024-07-31T14:30:52Z or 2024-07-31T14:30:52+02:00"""
    for fmt in ["%Y-%m-%dT%H:%M:%SZ", "%Y-%m-%dT%H:%M:%S%z", "%Y-%m-%dT%H:%M:%S.%fZ", "%Y-%m-%dT%H:%M:%S.%f%z"]:
        try:
            dt = datetime.strptime(ts, fmt)
            if dt.tzinfo is None:
                dt = dt.replace(tzinfo=timezone.utc)
            return dt.astimezone(timezone.utc)
        except ValueError:
            continue
    return None


def parse_unix_epoch(ts: str) -> Optional[datetime]:
    """Unix epoch: 1722447052 (seconds) or 1722447052000 (milliseconds)"""
    try:
        val = float(ts)
        if val > 1e12:  # milliseconds
            val = val / 1000.0
        return datetime.fromtimestamp(val, tz=timezone.utc)
    except (ValueError, OSError, OverflowError):
        return None


def parse_rfc2822(ts: str) -> Optional[datetime]:
    """RFC 2822: Wed, 31 Jul 2024 14:30:52 +0000"""
    from email.utils import parsedate_to_datetime
    try:
        dt = parsedate_to_datetime(ts)
        return dt.astimezone(timezone.utc)
    except (ValueError, TypeError):
        return None


def parse_http_date(ts: str) -> Optional[datetime]:
    """HTTP Date: Tue, 31 Jul 2024 14:30:52 GMT"""
    from email.utils import parsedate_to_datetime
    try:
        dt = parsedate_to_datetime(ts)
        return dt.astimezone(timezone.utc)
    except (ValueError, TypeError):
        return None


def parse_us_format(ts: str) -> Optional[datetime]:
    """US format: 07/31/2024 02:30 PM or 07/31/2024 14:30"""
    for fmt in ["%m/%d/%Y %I:%M %p", "%m/%d/%Y %H:%M", "%m/%d/%Y %I:%M:%S %p", "%m/%d/%Y %H:%M:%S"]:
        try:
            dt = datetime.strptime(ts.strip(), fmt)
            return dt.replace(tzinfo=timezone.utc)
        except ValueError:
            continue
    return None


def parse_eu_format(ts: str) -> Optional[datetime]:
    """EU format: 31.07.2024 14:30 or 31/07/2024 14:30"""
    for fmt in ["%d.%m.%Y %H:%M", "%d/%m/%Y %H:%M", "%d.%m.%Y %H:%M:%S", "%d/%m/%Y %H:%M:%S"]:
        try:
            dt = datetime.strptime(ts.strip(), fmt)
            return dt.replace(tzinfo=timezone.utc)
        except ValueError:
            continue
    return None


def parse_relative(ts: str) -> Optional[datetime]:
    """Relative: '3 hours ago', '2 days ago', '5 minutes ago', 'just now'"""
    ts_lower = ts.strip().lower()
    multipliers = {
        'second': 1, 'seconds': 1, 'sec': 1, 'secs': 1,
        'minute': 60, 'minutes': 60, 'min': 60, 'mins': 60,
        'hour': 3600, 'hours': 3600, 'hr': 3600, 'hrs': 3600,
        'day': 86400, 'days': 86400,
        'week': 604800, 'weeks': 604800, 'wk': 604800, 'wks': 604800,
        'month': 2592000, 'months': 2592000,
        'year': 31536000, 'years': 31536000,
    }
    if ts_lower in ('just now', 'now'):
        return NOW_UTC

    match = re.match(r'(\d+)\s+(\w+)\s+ago', ts_lower)
    if match:
        num = int(match.group(1))
        unit = match.group(2)
        if unit in multipliers:
            delta = timedelta(seconds=num * multipliers[unit])
            return NOW_UTC - delta

    # Handle "yesterday", "last monday", etc.
    day_names = ['monday', 'tuesday', 'wednesday', 'thursday', 'friday', 'saturday', 'sunday']
    if ts_lower == 'yesterday':
        return NOW_UTC - timedelta(days=1)
    if ts_lower.startswith('last '):
        day_name = ts_lower.replace('last ', '').strip()
        if day_name in day_names:
            target_weekday = day_names.index(day_name)
            current_weekday = NOW_UTC.weekday()
            days_back = (current_weekday - target_weekday) % 7
            if days_back == 0:
                days_back = 7
            return NOW_UTC - timedelta(days=days_back)

    return None


def parse_discord_snowflake(ts_str: str) -> Optional[datetime]:
    """Discord snowflake ID: extract timestamp from Discord message/ID"""
    try:
        snowflake = int(ts_str)
        if snowflake < 4194304:
            return None
        ms_epoch = (snowflake >> 22) + 1420070400000
        return datetime.fromtimestamp(ms_epoch / 1000.0, tz=timezone.utc)
    except (ValueError, OverflowError):
        return None


def parse_twitter_id(ts_str: str) -> Optional[datetime]:
    """Twitter/X snowflake ID: extract timestamp from tweet ID"""
    try:
        tweet_id = int(ts_str)
        if tweet_id < 1288834974657:
            return None
        ms_epoch = (tweet_id >> 22) + 1288834974657
        return datetime.fromtimestamp(ms_epoch / 1000.0, tz=timezone.utc)
    except (ValueError, OverflowError):
        return None


def parse_facebook_id(ts_str: str) -> Optional[datetime]:
    """Facebook ID: approximate timestamp from numeric ID"""
    try:
        fb_id = int(ts_str)
        if fb_id < 1290115200000:
            return None
        seconds = (fb_id - 1290115200000) / 1000.0
        return datetime.fromtimestamp(seconds, tz=timezone.utc)
    except (ValueError, OverflowError):
        return None


def parse_unix_ls_date(ts: str) -> Optional[datetime]:
    """Unix ls date: 'Jul 31 14:30' — assumes current year"""
    for fmt in ["%b %d %H:%M", "%b %d %Y"]:
        try:
            dt = datetime.strptime(ts.strip(), fmt)
            if dt.year == 1900:
                dt = dt.replace(year=NOW_UTC.year)
            return dt.replace(tzinfo=timezone.utc)
        except ValueError:
            continue
    return None


def parse_year_month_day(ts: str) -> Optional[datetime]:
    """Simple date: 2024-07-31 or 2024/07/31"""
    for fmt in ["%Y-%m-%d", "%Y/%m/%d"]:
        try:
            dt = datetime.strptime(ts.strip(), fmt)
            return dt.replace(tzinfo=timezone.utc)
        except ValueError:
            continue
    return None


# Parser chain — try each parser in order
PARSERS = [
    parse_iso8601,
    parse_unix_epoch,
    parse_rfc2822,
    parse_http_date,
    parse_us_format,
    parse_eu_format,
    parse_relative,
    parse_discord_snowflake,
    parse_twitter_id,
    parse_facebook_id,
    parse_unix_ls_date,
    parse_year_month_day,
]


def normalize_timestamp(ts_raw: str) -> Dict[str, Any]:
    """Try all parsers. Return result with validation."""
    result = {
        "raw": ts_raw,
        "normalized_utc": None,
        "valid": False,
        "validation_error": None,
        "parser_used": None,
    }

    for parser in PARSERS:
        dt = parser(ts_raw)
        if dt is not None:
            result["normalized_utc"] = dt.isoformat()
            result["parser_used"] = parser.__name__

            # VALIDATE: reject future timestamps
            if dt > NOW_UTC:
                result["valid"] = False
                result["validation_error"] = "FUTURE_TIMESTAMP"
            else:
                result["valid"] = True
            return result

    result["validation_error"] = "UNPARSABLE"
    return result


def process_events(input_path: str, output_path: str):
    """Read events JSON, normalize all timestamps, write results."""
    with open(input_path) as f:
        events = json.load(f)

    results = []
    stats = {"total": len(events), "valid": 0, "invalid": 0, "unparsable": 0}

    for event in events:
        ts_raw = event.get("timestamp") or event.get("date") or event.get("created_at") or ""
        norm = normalize_timestamp(str(ts_raw))
        combined = {**event, **norm}
        results.append(combined)
        if norm["valid"]:
            stats["valid"] += 1
        elif norm["validation_error"] == "UNPARSABLE":
            stats["unparsable"] += 1
        else:
            stats["invalid"] += 1

    output = {
        "agent": "osint-temporal",
        "step": "normalization",
        "stats": stats,
        "validation_time_utc": NOW_UTC.isoformat(),
        "events": results,
    }

    os.makedirs(os.path.dirname(output_path) if os.path.dirname(output_path) else ".", exist_ok=True)
    with open(output_path, "w") as f:
        json.dump(output, f, indent=2, default=str)
    print(json.dumps(stats))
    return output


if __name__ == "__main__":
    if len(sys.argv) < 3:
        print(json.dumps({"error": "Usage: python3 script.py <input.json> <output.json>"}))
        sys.exit(1)
    process_events(sys.argv[1], sys.argv[2])
```

## Gap Analysis Script

Save as `/tmp/osint_temporal_gaps.py`. Computes actual gap durations between events.

```python
#!/usr/bin/env python3
"""
OSINT Temporal — Gap analysis: detect digital silence periods.
Usage: python3 /tmp/osint_temporal_gaps.py <normalized_events.json> <output_gaps.json>
"""
import json
import sys
import os
from datetime import datetime, timezone, timedelta
from typing import List, Dict, Any


def compute_gaps(events: List[Dict[str, Any]]) -> Dict[str, Any]:
    """Compute gaps between consecutive events. Real computation, not estimates."""
    # Filter to valid events only
    valid_events = [e for e in events if e.get("valid") is True and e.get("normalized_utc")]

    if not valid_events:
        return {"error": "No valid timestamps to analyze", "total_events": len(events)}

    # Sort by timestamp
    valid_events.sort(key=lambda e: e["normalized_utc"])

    # Parse timestamps to datetime objects
    timestamps = []
    for e in valid_events:
        ts_str = e["normalized_utc"]
        # Handle various ISO formats
        ts_str = ts_str.replace("+00:00", "Z")
        if ts_str.endswith("Z"):
            ts_str = ts_str[:-1] + "+00:00"
        dt = datetime.fromisoformat(ts_str)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        timestamps.append((dt, e))

    # Compute gaps
    gaps = []
    for i in range(1, len(timestamps)):
        prev_dt, prev_event = timestamps[i - 1]
        curr_dt, curr_event = timestamps[i]
        delta = curr_dt - prev_dt
        delta_seconds = delta.total_seconds()
        delta_days = delta_seconds / 86400.0

        classification = "normal"
        if delta_days > 90:
            classification = "SIGNIFICANT_SILENCE"
        elif delta_days > 30:
            classification = "NOTABLE_GAP"
        elif delta_days > 14:
            classification = "extended_gap"

        gaps.append({
            "gap_index": i - 1,
            "start": prev_dt.isoformat(),
            "end": curr_dt.isoformat(),
            "duration_seconds": delta_seconds,
            "duration_hours": delta_seconds / 3600.0,
            "duration_days": round(delta_days, 2),
            "classification": classification,
            "prev_platform": prev_event.get("platform"),
            "curr_platform": curr_event.get("platform"),
            "prev_source": prev_event.get("source_url"),
            "curr_source": curr_event.get("source_url"),
        })

    significant = [g for g in gaps if g["classification"] == "SIGNIFICANT_SILENCE"]
    notable = [g for g in gaps if g["classification"] == "NOTABLE_GAP"]
    extended = [g for g in gaps if g["classification"] == "extended_gap"]

    # Compute gap statistics
    gap_days = [g["duration_days"] for g in gaps]
    gap_stats = {}
    if gap_days:
        gap_stats = {
            "count": len(gap_days),
            "mean_days": round(sum(gap_days) / len(gap_days), 2),
            "median_days": round(sorted(gap_days)[len(gap_days) // 2], 2),
            "min_days": round(min(gap_days), 2),
            "max_days": round(max(gap_days), 2),
            "std_dev_days": round(
                (sum((d - sum(gap_days) / len(gap_days)) ** 2 for d in gap_days) / len(gap_days)) ** 0.5,
                2
            ) if len(gap_days) > 1 else 0.0,
        }

    return {
        "agent": "osint-temporal",
        "step": "gap_analysis",
        "total_events_analyzed": len(valid_events),
        "total_gaps": len(gaps),
        "gap_statistics": gap_stats,
        "significant_silences": significant,
        "notable_gaps": notable,
        "extended_gaps": extended,
        "all_gaps": gaps,
    }


if __name__ == "__main__":
    if len(sys.argv) < 3:
        print(json.dumps({"error": "Usage: python3 script.py <input.json> <output.json>"}))
        sys.exit(1)
    with open(sys.argv[1]) as f:
        events = json.load(f)
    result = compute_gaps(events if isinstance(events, list) else events.get("events", []))
    os.makedirs(os.path.dirname(sys.argv[2]) if os.path.dirname(sys.argv[2]) else ".", exist_ok=True)
    with open(sys.argv[2], "w") as f:
        json.dump(result, f, indent=2, default=str)
    print(json.dumps({"gaps_found": len(result.get("all_gaps", [])), "significant": len(result.get("significant_silences", []))}))
```

## Frequency Analysis Script

Save as `/tmp/osint_temporal_frequency.py`. Computes real statistics: posts/day, std dev, burst detection, hourly distribution.

```python
#!/usr/bin/env python3
"""
OSINT Temporal — Frequency analysis: posting patterns, burst detection, bot indicators.
Usage: python3 /tmp/osint_temporal_frequency.py <normalized_events.json> <output_frequency.json>
"""
import json
import sys
import os
import math
from datetime import datetime, timezone, timedelta
from collections import Counter, defaultdict
from typing import List, Dict, Any


def compute_frequency(events: List[Dict[str, Any]]) -> Dict[str, Any]:
    """Compute actual frequency statistics. Every number is computed, not estimated."""
    valid_events = [e for e in events if e.get("valid") is True and e.get("normalized_utc")]

    if not valid_events:
        return {"error": "No valid timestamps to analyze", "total_events": len(events)}

    valid_events.sort(key=lambda e: e["normalized_utc"])

    # Parse timestamps
    datetimes = []
    for e in valid_events:
        ts_str = e["normalized_utc"].replace("+00:00", "Z")
        if ts_str.endswith("Z"):
            ts_str = ts_str[:-1] + "+00:00"
        dt = datetime.fromisoformat(ts_str)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        datetimes.append(dt)

    # Date range
    earliest = min(datetimes)
    latest = max(datetimes)
    total_span_days = max((latest - earliest).total_seconds() / 86400.0, 1.0)

    # Posts per day
    posts_per_day = len(valid_events) / total_span_days

    # Inter-post intervals (in seconds)
    intervals = []
    for i in range(1, len(datetimes)):
        delta = (datetimes[i] - datetimes[i - 1]).total_seconds()
        intervals.append(delta)

    # Interval statistics
    interval_stats = {}
    if intervals:
        mean_interval = sum(intervals) / len(intervals)
        variance = sum((iv - mean_interval) ** 2 for iv in intervals) / len(intervals)
        std_interval = math.sqrt(variance)
        interval_stats = {
            "count": len(intervals),
            "mean_seconds": round(mean_interval, 2),
            "std_dev_seconds": round(std_interval, 2),
            "min_seconds": round(min(intervals), 2),
            "max_seconds": round(max(intervals), 2),
            "median_seconds": round(sorted(intervals)[len(intervals) // 2], 2),
            "coefficient_of_variation": round(std_interval / mean_interval, 4) if mean_interval > 0 else 0.0,
        }

    # Hourly distribution (UTC hours 0-23)
    hour_counts = Counter(dt.hour for dt in datetimes)
    hour_distribution = {str(h): hour_counts.get(h, 0) for h in range(24)}
    peak_hours = sorted(hour_counts, key=hour_counts.get, reverse=True)[:5]
    quiet_hours = sorted(hour_counts, key=hour_counts.get)[:5]

    # Day-of-week distribution
    day_names = ["monday", "tuesday", "wednesday", "thursday", "friday", "saturday", "sunday"]
    day_counts = Counter(dt.weekday() for dt in datetimes)
    day_distribution = {day_names[d]: day_counts.get(d, 0) for d in range(7)}

    # Posts per day of week (normalize by number of that day in the date range)
    days_per_weekday = [0] * 7
    current = earliest
    while current <= latest:
        days_per_weekday[current.weekday()] += 1
        current += timedelta(days=1)
    posts_per_weekday_normalized = {}
    for d in range(7):
        denom = max(days_per_weekday[d], 1)
        posts_per_weekday_normalized[day_names[d]] = round(day_counts.get(d, 0) / denom, 2)

    # Burst detection: clusters of > 5 posts in 10 minutes
    bursts = []
    i = 0
    while i < len(datetimes):
        j = i + 1
        while j < len(datetimes) and (datetimes[j] - datetimes[i]).total_seconds() <= 600:  # 10 min window
            j += 1
        burst_count = j - i
        if burst_count >= 5:
            bursts.append({
                "burst_index": len(bursts),
                "start": datetimes[i].isoformat(),
                "end": datetimes[j - 1].isoformat(),
                "post_count": burst_count,
                "duration_seconds": (datetimes[j - 1] - datetimes[i]).total_seconds(),
            })
            i = j  # Skip past this burst
        else:
            i += 1

    # Sleep gap detection: find the longest consecutive period with zero activity
    hour_activity = set(dt.hour for dt in datetimes)
    sleep_candidates = []
    for start_hour in range(24):
        consecutive_quiet = 0
        h = start_hour
        while h % 24 not in hour_activity:
            consecutive_quiet += 1
            h += 1
            if consecutive_quiet >= 24:
                break
        if consecutive_quiet >= 5:
            sleep_candidates.append({
                "start_hour_utc": start_hour,
                "quiet_hours": consecutive_quiet,
            })
    sleep_gap = max(sleep_candidates, key=lambda x: x["quiet_hours"]) if sleep_candidates else None

    # Bot detection heuristics (real computation)
    bot_indicators = []
    human_indicators = []
    bot_score = 0.0

    # Check 1: Very regular posting (low CV of intervals)
    if intervals and interval_stats["coefficient_of_variation"] < 0.15:
        bot_indicators.append("very_regular_posting")
        bot_score += 0.3
    elif intervals and interval_stats["coefficient_of_variation"] > 0.5:
        human_indicators.append("natural_variance_in_timing")

    # Check 2: No sleep gap (24/7 activity)
    if sleep_gap is None or sleep_gap["quiet_hours"] < 4:
        bot_indicators.append("no_sleep_gap_24_7_activity")
        bot_score += 0.3
    elif sleep_gap and sleep_gap["quiet_hours"] >= 6:
        human_indicators.append(f"clear_sleep_gap_{sleep_gap['quiet_hours']}hrs")

    # Check 3: Excessive posting rate
    if posts_per_day > 100:
        bot_indicators.append(f"excessive_rate_{posts_per_day:.1f}_per_day")
        bot_score += 0.2

    # Check 4: Burst patterns
    if len(bursts) > 3:
        bot_indicators.append(f"multiple_bursts_{len(bursts)}_detected")
        bot_score += 0.2

    # Check 5: Weekday/weekend difference
    weekday_avg = sum(posts_per_weekday_normalized[d] for d in day_names[:5]) / 5
    weekend_avg = sum(posts_per_weekday_normalized[d] for d in day_names[5:]) / 2
    if weekday_avg > 0 and weekend_avg / weekday_avg < 0.6:
        human_indicators.append("weekend_activity_reduction")
    elif weekday_avg > 0 and abs(weekend_avg / weekday_avg - 1.0) < 0.1:
        bot_indicators.append("uniform_weekday_weekend_activity")
        bot_score += 0.1

    bot_probability = min(bot_score, 1.0)

    return {
        "agent": "osint-temporal",
        "step": "frequency_analysis",
        "total_events": len(valid_events),
        "date_range": {
            "earliest": earliest.isoformat(),
            "latest": latest.isoformat(),
            "span_days": round(total_span_days, 2),
        },
        "posts_per_day": round(posts_per_day, 4),
        "interval_statistics": interval_stats,
        "hour_distribution_utc": hour_distribution,
        "peak_hours_utc": peak_hours,
        "quiet_hours_utc": quiet_hours,
        "day_of_week_distribution": day_distribution,
        "posts_per_weekday_normalized": posts_per_weekday_normalized,
        "burst_clusters": bursts,
        "sleep_analysis": {
            "sleep_gap_detected": sleep_gap is not None,
            "longest_quiet_period": sleep_gap,
        },
        "bot_detection": {
            "bot_probability": round(bot_probability, 4),
            "bot_indicators": bot_indicators,
            "human_indicators": human_indicators,
        },
    }


if __name__ == "__main__":
    if len(sys.argv) < 3:
        print(json.dumps({"error": "Usage: python3 script.py <input.json> <output.json>"}))
        sys.exit(1)
    with open(sys.argv[1]) as f:
        data = json.load(f)
    events = data if isinstance(data, list) else data.get("events", [])
    result = compute_frequency(events)
    os.makedirs(os.path.dirname(sys.argv[2]) if os.path.dirname(sys.argv[2]) else ".", exist_ok=True)
    with open(sys.argv[2], "w") as f:
        json.dump(result, f, indent=2, default=str)
    print(json.dumps({"posts_per_day": result.get("posts_per_day"), "bot_probability": result.get("bot_detection", {}).get("bot_probability")}))
```

```python
            "step": "timezone_inference",
            "error": "INSUFFICIENT_DATA",
            "total_events": len(valid_events),
            "note": "Need at least 20 timestamped events for reliable timezone inference.",
            "inferred_timezone": None,
            "confidence": 0.0,
        }

    # Parse and collect UTC hours
    utc_hours = []
    for e in valid_events:
        ts_str = e["normalized_utc"].replace("+00:00", "Z")
        if ts_str.endswith("Z"):
            ts_str = ts_str[:-1] + "+00:00"
        dt = datetime.fromisoformat(ts_str)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        utc_hours.append(dt.hour)

    # Build UTC hour histogram
    hour_counts = Counter(utc_hours)
    total = len(utc_hours)

    # Find the weighted mean active hour in UTC
    weighted_sum = sum(h * count for h, count in hour_counts.items())
    mean_utc_hour = weighted_sum / total

    # Assume peak activity is around 9-11 AM local time
    # So the offset is approximately: mean_utc_hour - 10 (target local hour)
    # Round to nearest integer offset
    raw_offset = mean_utc_hour - 10.0
    # Normalize to [-12, +14] range
    while raw_offset > 14:
        raw_offset -= 24
    while raw_offset < -12:
        raw_offset += 24
    inferred_offset = round(raw_offset)

    # Calculate confidence based on how concentrated the activity is
    # Use standard deviation of hours (circular) — low std dev = high confidence
    # Convert hours to angles for circular statistics
    if len(utc_hours) > 1:
        sin_sum = sum(math.sin(2 * math.pi * h / 24) for h in utc_hours)
        cos_sum = sum(math.cos(2 * math.pi * h / 24) for h in utc_hours)
        R = math.sqrt(sin_sum ** 2 + cos_sum ** 2) / total  # Mean resultant length
        confidence = round(R, 4)  # 0 = uniform, 1 = all at same hour
    else:
        confidence = 0.0

    # Try each possible offset and score it
    # Higher score = more activity falls in 7AM-11PM local time window
    best_offset = inferred_offset
    best_score = 0
    for offset in range(-12, 15):
        local_hours = [(h + offset) % 24 for h in utc_hours]
        # Score: what fraction of activity falls in 7-23 local time
        active_in_waking = sum(1 for lh in local_hours if 7 <= lh <= 23)
        score = active_in_waking / total
        # Bonus for clear sleep gap (low activity 0-6 local)
        active_in_sleep = sum(1 for lh in local_hours if 0 <= lh <= 6)
        sleep_ratio = active_in_sleep / total
        if sleep_ratio < 0.1:
            score += 0.1  # Bonus for clear sleep gap
        if score > best_score:
            best_score = score
            best_offset = offset

    # Format the offset as UTC+X or UTC-X
    if best_offset >= 0:
        tz_label = f"UTC+{best_offset}"
    else:
        tz_label = f"UTC{best_offset}"

    # Per-platform analysis
    platform_hours = {}
    for e in valid_events:
        platform = e.get("platform", "unknown")
        ts_str = e["normalized_utc"].replace("+00:00", "Z")
        if ts_str.endswith("Z"):
            ts_str = ts_str[:-1] + "+00:00"
        dt = datetime.fromisoformat(ts_str)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        if platform not in platform_hours:
            platform_hours[platform] = []
        platform_hours[platform].append(dt.hour)

    platform_analysis = {}
    for platform, hours in platform_hours.items():
        ph_counts = Counter(hours)
        ph_mean = sum(h * c for h, c in ph_counts.items()) / len(hours)
        p_offset = round(ph_mean - 10.0)
        while p_offset > 14:
            p_offset -= 24
        while p_offset < -12:
            p_offset += 24
        platform_analysis[platform] = {
            "event_count": len(hours),
            "mean_utc_hour": round(ph_mean, 2),
            "inferred_offset": p_offset,
        }

    return {
        "agent": "osint-temporal",
        "step": "timezone_inference",
        "total_events": len(valid_events),
        "inferred_timezone": tz_label,
        "inferred_offset_hours": best_offset,
        "confidence": round(min(confidence, 1.0), 4),
        "utc_hour_distribution": {str(h): hour_counts.get(h, 0) for h in range(24)},
        "mean_utc_hour": round(mean_utc_hour, 2),
        "best_fit_score": round(best_score, 4),
        "platform_breakdown": platform_analysis,
    }


if __name__ == "__main__":
    if len(sys.argv) < 3:
```

## Error Handling

- If a timestamp cannot be parsed, keep the original string, tag it `UNPARSABLE`, and include it in the invalid_timestamps list
- If a timestamp is in the future, tag it `FUTURE_TIMESTAMP` and EXCLUDE it from all analysis
- Never fail the entire analysis due to malformed timestamps
- If a Python script fails, capture the traceback and include it in the output under `script_errors`

## Data Quality

- Minimum 20 timestamped events needed for reliable frequency analysis
- Minimum 50 events for reliable timezone inference
- If below these thresholds, set `data_quality: "LOW"` and reduce all confidence scores by 0.3
- Always report the total number of events and how many were filtered out
