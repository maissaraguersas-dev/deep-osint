from datetime import datetime, timezone

# Replace with actual timestamps from Step 1
timestamps_str = [
    # "2024-01-15T10:30:00Z",
    # Add all collected timestamps here
]

if len(timestamps_str) < 3:
    print("TEMPORAL_ANALYSIS: NOT_EXECUTED — fewer than 3 timestamps")
else:
    ts = [datetime.fromisoformat(t.replace('Z', '+00:00')) for t in timestamps_str]
    ts.sort()
    intervals = [(ts[i+1] - ts[i]).total_seconds() for i in range(len(ts)-1)]

    mean_iv = statistics.mean(intervals)
    median_iv = statistics.median(intervals)
    std_iv = statistics.stdev(intervals) if len(intervals) > 1 else 0
    cv = std_iv / mean_iv if mean_iv > 0 else float('inf')

    print(f"N posts: {len(ts)}")
    print(f"Time span: {(ts[-1] - ts[0]).days} days")
    print(f"Mean interval: {mean_iv:.1f}s ({mean_iv/3600:.2f}h)")
    print(f"Median interval: {median_iv:.1f}s ({median_iv/3600:.2f}h)")
    print(f"Std dev: {std_iv:.1f}s ({std_iv/3600:.2f}h)")
    print(f"CV: {cv:.4f}")
    print(f"Min interval: {min(intervals):.1f}s")
    print(f"Max interval: {max(intervals):.1f}s")

    # Flag near-impossible intervals
    fast = [iv for iv in intervals if iv < 10]
    if fast:
        print(f"FLAG: {len(fast)} intervals < 10s (near-impossible for humans)")

    # Flag suspiciously regular intervals
    if cv < 0.05 and len(intervals) > 5:
        print(f"FLAG: CV={cv:.4f} — posts are suspiciously regular (automation?)")

    # Check for exact automation multiples
    for multiple in [60, 300, 3600, 86400]:
        matches = sum(1 for iv in intervals if abs(iv - multiple) < 5)
        if matches > len(intervals) * 0.3:
            print(f"FLAG: {matches}/{len(intervals)} intervals ~= {multiple}s (cron pattern)")

    # Posting hour distribution
    hours = [t.hour for t in ts]
    hour_dist = {h: hours.count(h) for h in range(24) if hours.count(h) > 0}
    print(f"Posting hours (UTC): {hour_dist}")

    # Weekend/weekday
    weekends = sum(1 for t in ts if t.weekday() >= 5)
    weekdays = len(ts) - weekends
    print(f"Weekday posts: {weekdays} ({weekdays/len(ts)*100:.1f}%)")
    print(f"Weekend posts: {weekends} ({weekends/len(ts)*100:.1f}%)")

#!/usr/bin/env python3
"""
OSINT Account Classification — Probability Normalizer
Ensures all probabilities sum to exactly 1.0 via softmax.
"""
import math
import json
from datetime import datetime, timezone

# ============================================================
# STEP A: Record raw evidence scores (0.0 to 10.0) for each check
# These MUST be based on ACTUAL executed checks, not guesses.
# ============================================================

raw_scores = {
    # Profile Homogeneity (Step 2)
    "profile_homogeneity_score": 0.0,  # 0=fully custom, 10=all template indicators

    "temporal_checks_executed": False,

    "persona_checks_consistent": 0,
    "persona_checks_inconsistent": 0,
    "persona_checks_total_assessed": 0,

    # Content Analysis (Step 6)
    "content_anomaly_score": 0.0,  # 0=organic, 10=automated/suspicious
    "content_checks_executed": False,
}

# ============================================================
# STEP B: Compute aggregate anomaly score
# ============================================================

weights = {
    "profile_homogeneity_score": 0.20,
    "temporal_anomaly_score": 0.25,
    "network_anomaly_score": 0.25,
    "persona_incoherence_score": 0.20,
    "content_anomaly_score": 0.10,
}

weighted_anomaly = sum(
    raw_scores[k] * weights[k] for k in weights
)
print(f"Weighted Anomaly Score: {weighted_anomaly:.4f} (0=clean, 10=synthetic)")

# ============================================================
# STEP C: Compute raw classification logits
# Higher anomaly = more synthetic classifications get higher logits
# ============================================================

def compute_logits(anomaly):
    """Convert anomaly score to classification logits."""
    return {
        "genuine_individual":     max(0.1, 10.0 - anomaly * 1.2),
        "reseller_operation":     max(0.1, 2.0 + anomaly * 0.3),
        "coordinated_farm":       max(0.1, anomaly * 0.8),
        "semi_automated":         max(0.1, anomaly * 0.6),
        "fully_synthetic":        max(0.1, anomaly * 0.5 - 1.0),
        "persona_larp":           max(0.1, 1.5 + anomaly * 0.4),
    }

logits = compute_logits(weighted_anomaly)
print(f"\nRaw logits: {json.dumps(logits, indent=2)}")

# ============================================================
# STEP D: Softmax normalization to ensure sum = exactly 1.0
# ============================================================

def softmax(logits_dict):
    """Compute softmax probabilities. Guaranteed to sum to 1.0."""
    keys = list(logits_dict.keys())
    vals = [logits_dict[k] for k in keys]
    max_val = max(vals)
    exp_vals = [math.exp(v - max_val) for v in vals]
    total = sum(exp_vals)
    probs = {k: exp_vals[i] / total for i, k in enumerate(keys)}
    return probs

probabilities = softmax(logits)

# ============================================================
# STEP E: Consistency check (Rule 0.4)
# ============================================================

prob_sum = sum(probabilities.values())
print(f"\nProbability sum: {prob_sum:.10f}")
assert abs(prob_sum - 1.0) < 1e-9, f"PROBABILITY SUM ERROR: {prob_sum}"
print("✓ Probabilities sum to exactly 1.0")

# Top classification
top_class = max(probabilities, key=probabilities.get)
top_prob = probabilities[top_class]
print(f"\nTop classification: {top_class} ({top_prob:.4f})")
print(f"\nFull classification:")
for cls, prob in sorted(probabilities.items(), key=lambda x: -x[1]):
    bar = "█" * int(prob * 40)
    print(f"  {cls:25s} {prob:.6f} {bar}")

# ============================================================
# STEP F: Qualitative-quantitative consistency check
# ============================================================

# Check: if anomaly is high (>7), fully_synthetic should not be lowest
synthetic_categories = ["coordinated_farm", "fully_synthetic", "semi_automated"]
synthetic_total = sum(probabilities[c] for c in synthetic_categories)
if weighted_anomaly > 7.0 and synthetic_total < 0.3:
    print("\n⚠️  CONSISTENCY WARNING: High anomaly score but low synthetic probability.")
    print("   This suggests a miscalibration. Review raw scores.")

if weighted_anomaly < 3.0 and probabilities["genuine_individual"] < 0.5:
    print("\n⚠️  CONSISTENCY WARNING: Low anomaly score but low genuine probability.")
    print("   This suggests a miscalibration. Review raw scores.")

# ============================================================
# STEP G: Timestamp validation (Rule 0.5)
# ============================================================

now = datetime.now(timezone.utc)
print(f"\nAnalysis timestamp: {now.isoformat()}")
print("All timestamps in this analysis must be <= the above time.")

# ============================================================
# STEP H: Execution coverage report
# ============================================================

total_checks = 15 + 7 + 5 + 18 + 7  # Steps 2-6
executed = (
    raw_scores["homogeneity_checks_total_assessed"] +
    (1 if raw_scores["temporal_checks_executed"] else 0) +
    (1 if raw_scores["network_checks_executed"] else 0) +
    raw_scores["persona_checks_total_assessed"] +
    (1 if raw_scores["content_checks_executed"] else 0)
)
print(f"\nExecution coverage: {executed}/{total_checks} check groups executed")
if executed < total_checks * 0.5:
    print("⚠️  WARNING: Less than 50% of checks executed. Verdict confidence is LOW.")

# ============================================================
# STEP I: Authenticity score
# ============================================================

# Weighted average: higher = more likely authentic
authenticity_components = {
    "profile_uniqueness": 1.0 - (raw_scores["profile_homogeneity_score"] / 10.0),
    "temporal_naturalness": 1.0 - (raw_scores["temporal_anomaly_score"] / 10.0),
    "network_organic": 1.0 - (raw_scores["network_anomaly_score"] / 10.0),
    "persona_coherence": 1.0 - (raw_scores["persona_incoherence_score"] / 10.0),
    "content_organic": 1.0 - (raw_scores["content_anomaly_score"] / 10.0),
}

auth_weights = {
    "profile_uniqueness": 0.15,
    "temporal_naturalness": 0.25,
    "network_organic": 0.25,
    "persona_coherence": 0.25,
    "content_organic": 0.10,
}

authenticity_score = sum(
    authenticity_components[k] * auth_weights[k] for k in auth_weights
)
print(f"\n{'='*60}")
print(f"AUTHENTICITY SCORE: {authenticity_score:.4f} (0.0=fake, 1.0=genuine)")
print(f"{'='*60}")

if authenticity_score >= 0.8:
    rating = "LIKELY GENUINE"
elif authenticity_score >= 0.6:
    rating = "PROBABLY GENUINE"
elif authenticity_score >= 0.4:
    rating = "UNCERTAIN"
elif authenticity_score >= 0.2:
    rating = "PROBABLY INAUTHENTIC"
else:
    rating = "LIKELY INAUTHENTIC"

print(f"Rating: {rating}")

# ============================================================
# STEP J: Output machine-readable JSON
# ============================================================

result = {
    "analysis_timestamp": now.isoformat(),
    "authenticity_score": round(authenticity_score, 4),
    "authenticity_rating": rating,
    "weighted_anomaly_score": round(weighted_anomaly, 4),
    "classification_probabilities": {k: round(v, 6) for k, v in probabilities.items()},
    "probability_sum": round(sum(probabilities.values()), 10),
    "top_classification": top_class,
    "top_classification_confidence": round(top_prob, 4),
    "execution_coverage": {
        "executed_check_groups": executed,
        "total_check_groups": total_checks,
        "coverage_percentage": round(executed / total_checks * 100, 1),
    },
    "raw_scores": {k: v for k, v in raw_scores.items()},
    "consistency_warnings": [],
}

# Add consistency warnings
if weighted_anomaly > 7.0 and synthetic_total < 0.3:
    result["consistency_warnings"].append(
        "High anomaly but low synthetic probability"
    )
if weighted_anomaly < 3.0 and probabilities["genuine_individual"] < 0.5:
    result["consistency_warnings"].append(
        "Low anomaly but low genuine probability"
    )

print(f"\nJSON Output:")
print(json.dumps(result, indent=2))
