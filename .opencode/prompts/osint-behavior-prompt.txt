```python
#!/usr/bin/env python3
"""
OSINT Behavior — Stylometric feature extraction.
Computes: TTR, hapax legomena, avg word length, function word frequencies,
sentence length stats, punctuation frequencies, contraction rate, emoji frequency.
Usage: python3 /tmp/osint_behavior_features.py <input_texts.json> <output_features.json>
Input: JSON file with dict of {"account_id": "full text here...", ...}
Output: JSON file with all computed features per account
"""
import json
import sys
import os
import re
import math
import unicodedata
from collections import Counter
from typing import Dict, Any, List, Tuple


# --- Text Cleaning and Tokenization ---

def extract_emojis(text: str) -> List[str]:
    """Extract emoji characters from text."""
    emojis = []
    for char in text:
        if ord(char) > 0x1F000 or (ord(char) > 0x2600 and ord(char) < 0x27BF):
            emojis.append(char)
    return emojis


def tokenize_words(text: str) -> List[str]:
    """Tokenize text into words, preserving contractions as single tokens."""
    # Remove emojis for word tokenization
    clean = re.sub(r'[\U0001F600-\U0001F64F\U0001F300-\U0001F5FF\U0001F680-\U0001F6FF\U0001F1E0-\U0001F1FF\U00002702-\U000027B0\U0000FE00-\U0000FE0F\U0001F900-\U0001F9FF\U0001FA00-\U0001FA6F\U0001FA70-\U0001FAFF\U00002700-\U000027BF]', '', text)
    words = re.findall(r"[a-zA-Z]+(?:'[a-zA-Z]+)*|[0-9]+(?:\.[0-9]+)?", clean.lower())
    return [w for w in words if len(w) > 0]


def tokenize_sentences(text: str) -> List[str]:
    """Split text into sentences."""
    sentences = re.split(r'[.!?]+', text)
    return [s.strip() for s in sentences if len(s.strip()) > 5]  # Min 5 chars


# --- Lexical Features ---

def compute_type_token_ratio(words: List[str]) -> float:
    """Unique words / total words. Measures vocabulary richness."""
    if not words:
        return 0.0
    return len(set(words)) / len(words)


def compute_hapax_legomena(words: List[str]) -> Tuple[float, int, int]:
    """Words used exactly once / total words. Returns (ratio, count, total)."""
    if not words:
        return 0.0, 0, 0
    counts = Counter(words)
    hapax = sum(1 for w, c in counts.items() if c == 1)
    return hapax / len(words), hapax, len(words)


def compute_avg_word_length(words: List[str]) -> Dict[str, float]:
    """Average word length and distribution stats."""
    if not words:
        return {"mean": 0.0, "std": 0.0, "min": 0, "max": 0, "median": 0}
    lengths = [len(w) for w in words]
    mean = sum(lengths) / len(lengths)
    variance = sum((l - mean) ** 2 for l in lengths) / len(lengths)
    std = math.sqrt(variance)
    sorted_lengths = sorted(lengths)
    return {
        "mean": round(mean, 4),
        "std": round(std, 4),
        "min": min(lengths),
        "max": max(lengths),
        "median": sorted_lengths[len(sorted_lengths) // 2],
    }


def compute_function_word_frequencies(words: List[str]) -> Dict[str, float]:
    """Frequency of top 50 English function words relative to total word count."""
    FUNCTION_WORDS = [
        "the", "be", "to", "of", "and", "a", "in", "that", "have", "i",
        "it", "for", "not", "on", "with", "he", "as", "you", "do", "at",
        "this", "but", "his", "by", "from", "they", "we", "say", "her", "she",
        "or", "an", "will", "my", "one", "all", "would", "there", "their", "what",
        "so", "up", "out", "if", "about", "who", "get", "which", "go", "me",
    ]
    if not words:
        return {fw: 0.0 for fw in FUNCTION_WORDS}
    counts = Counter(words)
    total = len(words)
    return {fw: round(counts.get(fw, 0) / total, 6) for fw in FUNCTION_WORDS}


# --- Syntactic Features ---

def compute_sentence_features(sentences: List[str], words: List[str]) -> Dict[str, Any]:
    """Sentence length statistics."""
    if not sentences:
        return {
            "count": 0, "avg_length_words": 0.0, "std_length_words": 0.0,
            "min_length": 0, "max_length": 0, "median_length": 0,
        }
    sent_word_counts = [len(tokenize_words(s)) for s in sentences]
    sent_word_counts = [c for c in sent_word_counts if c > 0]
    if not sent_word_counts:
        return {"count": len(sentences), "avg_length_words": 0.0, "std_length_words": 0.0, "min_length": 0, "max_length": 0, "median_length": 0}

    mean = sum(sent_word_counts) / len(sent_word_counts)
    variance = sum((c - mean) ** 2 for c in sent_word_counts) / len(sent_word_counts)
    std = math.sqrt(variance)
    sorted_counts = sorted(sent_word_counts)

    return {
        "count": len(sent_word_counts),
        "avg_length_words": round(mean, 4),
        "std_length_words": round(std, 4),
        "min_length": min(sent_word_counts),
        "max_length": max(sent_word_counts),
        "median_length": sorted_counts[len(sorted_counts) // 2],
    }


def compute_punctuation_features(text: str) -> Dict[str, float]:
    """Frequency of each punctuation mark per 1000 characters."""
    if not text:
        return {}
    chars = len(text)
    punctuation = [
        (".", "period"), (",", "comma"), (";", "semicolon"),
        (":", "colon"), ("!", "exclamation"), ("?", "question"),
        ("-", "hyphen"), ("—", "emdash"), ("–", "endash"),
        ("(", "open_paren"), (")", "close_paren"),
        ("'", "single_quote"), ('"', "double_quote"),
    ]
    result = {}
    total_punct = 0
    for char, name in punctuation:
        count = text.count(char)
        result[name] = round(count / chars * 1000, 4)
        total_punct += count
    result["total_punctuation_per_1000_chars"] = round(total_punct / chars * 1000, 4)
    return result


def compute_contraction_rate(words: List[str]) -> Dict[str, Any]:
    """Rate of contraction usage (don't, can't, etc.)"""
    CONTRACTIONS = {
        "don't", "doesn't", "didn't", "can't", "cannot", "couldn't", "won't",
        "wouldn't", "shouldn't", "isn't", "aren't", "wasn't", "weren't",
        "hasn't", "haven't", "hadn't", "i'm", "you're", "he's", "she's",
        "it's", "we're", "they're", "i've", "you've", "we've", "they've",
        "i'll", "you'll", "he'll", "she'll", "we'll", "they'll",
        "i'd", "you'd", "he'd", "she'd", "we'd", "they'd",
        "let's", "that's", "who's", "what's", "there's", "here's",
    }
    if not words:
        return {"contraction_rate": 0.0, "contraction_count": 0, "total_words": 0}
    contraction_count = sum(1 for w in words if w in CONTRACTIONS)
    return {
        "contraction_rate": round(contraction_count / len(words), 6),
        "contraction_count": contraction_count,
        "total_words": len(words),
    }


def compute_emoji_features(text: str, words: List[str]) -> Dict[str, Any]:
    """Emoji usage frequency and top emojis."""
    emojis = extract_emojis(text)
    word_count = max(len(words), 1)
    emoji_counts = Counter(emojis)
    top_emojis = emoji_counts.most_common(10)
    return {
        "emoji_count": len(emojis),
        "emoji_frequency_per_100_words": round(len(emojis) / word_count * 100, 4),
        "unique_emoji_count": len(emoji_counts),
        "top_emojis": [{"emoji": e, "count": c} for e, c in top_emojis],
    }


# --- Bot Detection: Shannon Entropy ---

def compute_shannon_entropy(text: str) -> Dict[str, Any]:
    """Compute Shannon entropy of character distribution.
    Natural English text: ~4.0-4.5 bits/char.
    Repetitive/bot text: < 3.5 or suspiciously > 4.8.
    """
    if not text:
        return {"entropy_bits_per_char": 0.0, "unique_chars": 0, "total_chars": 0}
    char_counts = Counter(text)
    total = len(text)
    entropy = 0.0
    for count in char_counts.values():
        if count > 0:
            p = count / total
            entropy -= p * math.log2(p)
    return {
        "entropy_bits_per_char": round(entropy, 4),
        "unique_chars": len(char_counts),
        "total_chars": total,
        "assessment": (
            "LOW_ENTROPY_BOT_LIKELY" if entropy < 3.5
            else "SUSPICIOUSLY_HIGH" if entropy > 4.8
            else "NORMAL_RANGE"
        ),
    }


# --- Bot Detection: Repetition with Edit Distance ---

def levenshtein_distance(s1: str, s2: str) -> int:
    """Compute Levenshtein edit distance between two strings."""
    if len(s1) < len(s2):
        return levenshtein_distance(s2, s1)
    if len(s2) == 0:
        return len(s1)
    previous_row = list(range(len(s2) + 1))
    for i, c1 in enumerate(s1):
        current_row = [i + 1]
        for j, c2 in enumerate(s2):
            insertions = previous_row[j + 1] + 1
            deletions = current_row[j] + 1
            substitutions = previous_row[j] + (c1 != c2)
            current_row.append(min(insertions, deletions, substitutions))
        previous_row = current_row
    return previous_row[-1]


def detect_repetitions(posts: List[str], edit_distance_threshold: float = 0.2) -> Dict[str, Any]:
    """Detect near-duplicate posts using edit distance.
    Flags posts where edit distance / max(len1, len2) < threshold.
    """
    repetitions = []
    exact_repetitions = []
    post_counts = Counter(posts)

    # Exact repetitions
    for post, count in post_counts.items():
        if count > 1:
            exact_repetitions.append({"text_preview": post[:200], "count": count})

    # Near-duplicates (edit distance based)
    checked = set()
    for i in range(len(posts)):
        for j in range(i + 1, len(posts)):
            pair_key = (i, j)
            if pair_key in checked:
                continue
            checked.add(pair_key)
            if len(posts[i]) < 20 or len(posts[j]) < 20:
                continue
            dist = levenshtein_distance(posts[i], posts[j])
            max_len = max(len(posts[i]), len(posts[j]))
            ratio = dist / max_len if max_len > 0 else 1.0
            if ratio < edit_distance_threshold:
                repetitions.append({
                    "post_a_index": i,
                    "post_b_index": j,
                    "edit_distance": dist,
                    "max_length": max_len,
                    "distance_ratio": round(ratio, 4),
                    "preview_a": posts[i][:200],
                    "preview_b": posts[j][:200],
                })

    return {
        "exact_repetitions": exact_repetitions,
        "exact_repetition_count": len(exact_repetitions),
        "near_duplicates": repetitions,
        "near_duplicate_count": len(repetitions),
        "is_suspicious": len(exact_repetitions) > 3 or len(repetitions) > 5,
    }


# --- Main Feature Extraction ---

def extract_all_features(account_id: str, text: str, posts: List[str] = None) -> Dict[str, Any]:
    """Extract all stylometric features for a single account."""
    words = tokenize_words(text)
    sentences = tokenize_sentences(text)
    word_count = len(words)

    # Data quality check
    if word_count < 100:
        return {
            "account_id": account_id,
            "status": "INSUFFICIENT_DATA",
            "word_count": word_count,
            "note": f"Only {word_count} words. Need at least 100 for reliable analysis.",
            "confidence": 0.3,
        }

    ttr = compute_type_token_ratio(words)
    hapax_ratio, hapax_count, _ = compute_hapax_legomena(words)
    avg_wl = compute_avg_word_length(words)
    func_words = compute_function_word_frequencies(words)
    sent_feats = compute_sentence_features(sentences, words)
    punct_feats = compute_punctuation_features(text)
    contractions = compute_contraction_rate(words)
    emoji_feats = compute_emoji_features(text, words)
    entropy = compute_shannon_entropy(text)

    # Repetition detection (if we have individual posts)
    repetition = {"exact_repetitions": [], "near_duplicates": [], "exact_repetition_count": 0, "near_duplicate_count": 0, "is_suspicious": False}
    if posts and len(posts) > 1:
        repetition = detect_repetitions(posts)

    # Build feature vector for cosine similarity (numeric only, normalized)
    feature_vector = {
        "type_token_ratio": ttr,
        "hapax_legomena_ratio": hapax_ratio,
        "avg_word_length": avg_wl["mean"],
        "sentence_length_mean": sent_feats["avg_length_words"],
        "sentence_length_std": sent_feats["std_length_words"],
        "contraction_rate": contractions["contraction_rate"],
        "emoji_frequency": emoji_feats["emoji_frequency_per_100_words"],
        "punctuation_total": punct_feats.get("total_punctuation_per_1000_chars", 0),
        "period_freq": punct_feats.get("period", 0),
        "comma_freq": punct_feats.get("comma", 0),
        "exclamation_freq": punct_feats.get("exclamation", 0),
        "question_freq": punct_feats.get("question", 0),
        "entropy": entropy["entropy_bits_per_char"],
    }

    return {
        "account_id": account_id,
        "status": "ANALYZED",
        "word_count": word_count,
        "post_count": len(posts) if posts else 1,
        "sentence_count": sent_feats["count"],
        "confidence": 1.0 if word_count >= 500 else 0.7,
        "lexical": {
            "type_token_ratio": round(ttr, 6),
            "hapax_legomena_ratio": round(hapax_ratio, 6),
            "hapax_legomena_count": hapax_count,
            "avg_word_length": avg_wl,
            "function_word_frequencies": func_words,
        },
        "syntactic": {
            "sentence_features": sent_feats,
            "punctuation": punct_feats,
            "contraction_rate": contractions["contraction_rate"],
            "contraction_count": contractions["contraction_count"],
        },
        "orthographic": {
            "emoji_features": emoji_feats,
        },
        "bot_detection": {
            "shannon_entropy": entropy,
            "repetition_analysis": repetition,
            "bot_probability": 0.0,  # Computed after all features
        },
        "feature_vector": feature_vector,
    }


def main(input_path: str, output_path: str):
    """Read input, extract features for all accounts, write output."""
    with open(input_path) as f:
        data = json.load(f)

    # data should be {"account_id": "full text", ...} or {"account_id": {"text": "...", "posts": [...]}}
    results = {}
    for account_id, value in data.items():
        if isinstance(value, dict):
            text = value.get("text", "")
            posts = value.get("posts", [text])
        else:
            text = str(value)
            posts = [text]
        features = extract_all_features(account_id, text, posts)
        results[account_id] = features

    # Compute bot probability for each account
    for account_id, feat in results.items():
        if feat.get("status") != "ANALYZED":
            continue
        bot_score = 0.0
        entropy = feat["bot_detection"]["shannon_entropy"]["entropy_bits_per_char"]
        if entropy < 3.5:
            bot_score += 0.3
        if feat["bot_detection"]["repetition_analysis"].get("is_suspicious"):
            bot_score += 0.4
        ttr = feat["lexical"]["type_token_ratio"]
        if ttr < 0.25:
            bot_score += 0.2
        sent_std = feat["syntactic"]["sentence_features"]["std_length_words"]
        if sent_std < 2.0 and feat["syntactic"]["sentence_features"]["count"] > 10:
            bot_score += 0.1
        feat["bot_detection"]["bot_probability"] = round(min(bot_score, 1.0), 4)

    output = {"agent": "osint-behavior", "step": "feature_extraction", "accounts": results}
    os.makedirs(os.path.dirname(output_path) if os.path.dirname(output_path) else ".", exist_ok=True)
    with open(output_path, "w") as f:
        json.dump(output, f, indent=2, default=str)
    print(json.dumps({"accounts_analyzed": len(results), "insufficient_data": sum(1 for v in results.values() if v.get("status") == "INSUFFICIENT_DATA")}))


if __name__ == "__main__":
    if len(sys.argv) < 3:
        print(json.dumps({"error": "Usage: python3 script.py <input.json> <output.json>"}))
        sys.exit(1)
    main(sys.argv[1], sys.argv[2])
```

## Cosine Similarity Script

Save this as `/tmp/osint_behavior_similarity.py`. Computes ACTUAL cosine similarity between feature vectors. Not estimates.

```python
#!/usr/bin/env python3
"""
OSINT Behavior — Cosine similarity between stylometric feature vectors.
Usage: python3 /tmp/osint_behavior_similarity.py <features.json> <output_similarity.json>
Input: Output from osint_behavior_features.py
Output: Pairwise cosine similarity scores with interpretation
"""
import json
import sys
import os
import math
from typing import Dict, Any, List, Tuple
from itertools import combinations


def cosine_similarity(vec_a: Dict[str, float], vec_b: Dict[str, float]) -> Tuple[float, Dict[str, Any]]:
    """Compute cosine similarity between two feature vectors (dicts).
    Returns (similarity_score, details).
    Formula: cos(θ) = (A · B) / (||A|| × ||B||)
    """
    # Get common keys (intersection of both vectors)
    all_keys = set(vec_a.keys()) | set(vec_b.keys())

    dot_product = 0.0
    norm_a = 0.0
    norm_b = 0.0
    feature_contributions = {}

    for key in all_keys:
        a_val = vec_a.get(key, 0.0)
        b_val = vec_b.get(key, 0.0)
        dot_product += a_val * b_val
        norm_a += a_val ** 2
        norm_b += b_val ** 2
        feature_contributions[key] = {
            "a": round(a_val, 6),
            "b": round(b_val, 6),
            "product": round(a_val * b_val, 8),
        }

    norm_a = math.sqrt(norm_a)
    norm_b = math.sqrt(norm_b)

    if norm_a == 0 or norm_b == 0:
        return 0.0, {"error": "Zero-norm vector", "norm_a": norm_a, "norm_b": norm_b}

    similarity = dot_product / (norm_a * norm_b)

    # Feature-level comparison
    matching_features = []
    diverging_features = []
    for key in all_keys:
        a_val = vec_a.get(key, 0.0)
        b_val = vec_b.get(key, 0.0)
        max_val = max(abs(a_val), abs(b_val), 0.0001)
        diff_ratio = abs(a_val - b_val) / max_val
        if diff_ratio < 0.2:
            matching_features.append({"feature": key, "a": round(a_val, 6), "b": round(b_val, 6), "diff_ratio": round(diff_ratio, 4)})
        elif diff_ratio > 0.5:
            diverging_features.append({"feature": key, "a": round(a_val, 6), "b": round(b_val, 6), "diff_ratio": round(diff_ratio, 4)})

    details = {
        "dot_product": round(dot_product, 8),
        "norm_a": round(norm_a, 8),
        "norm_b": round(norm_b, 8),
        "matching_features": matching_features,
        "diverging_features": diverging_features,
        "feature_count": len(all_keys),
    }

    return round(similarity, 6), details


def interpret_similarity(score: float) -> str:
    """Interpret cosine similarity score."""
    if score >= 0.90:
        return "EXTREMELY_LIKELY_SAME_AUTHOR"
    elif score >= 0.75:
        return "LIKELY_SAME_AUTHOR"
    elif score >= 0.60:
        return "POSSIBLE_SAME_AUTHOR_INCONCLUSIVE"
    elif score >= 0.40:
        return "DIFFERENT_AUTHORS_OR_OBFUSCATED"
    else:
        return "ALMOST_CERTAINLY_DIFFERENT_AUTHORS"


def main(input_path: str, output_path: str):
    """Read features, compute pairwise cosine similarity, write output."""
    with open(input_path) as f:
        data = json.load(f)

    accounts = data.get("accounts", {})
    # Filter to only accounts with feature vectors
    analyzable = {k: v for k, v in accounts.items() if v.get("status") == "ANALYZED" and "feature_vector" in v}
    account_ids = list(analyzable.keys())

    comparisons = []
    for a_id, b_id in combinations(account_ids, 2):
        vec_a = analyzable[a_id]["feature_vector"]
        vec_b = analyzable[b_id]["feature_vector"]

        similarity, details = cosine_similarity(vec_a, vec_b)
        interpretation = interpret_similarity(similarity)

        # Adjust confidence based on data quality
        words_a = analyzable[a_id].get("word_count", 0)
        words_b = analyzable[b_id].get("word_count", 0)
        min_words = min(words_a, words_b)

        confidence = 1.0
        if min_words < 200:
            confidence = 0.5
        elif min_words < 500:
            confidence = 0.7

        comparison = {
            "account_a": a_id,
            "account_b": b_id,
            "cosine_similarity": similarity,
            "interpretation": interpretation,
            "confidence": round(confidence, 2),
            "words_a": words_a,
            "words_b": words_b,
            "matching_features": [f["feature"] for f in details.get("matching_features", [])],
            "diverging_features": [f["feature"] for f in details.get("diverging_features", [])],
            "math_details": details,
        }
        comparisons.append(comparison)

    # Sort by similarity descending
    comparisons.sort(key=lambda c: c["cosine_similarity"], reverse=True)

    output = {
        "agent": "osint-behavior",
        "step": "cosine_similarity",
        "total_comparisons": len(comparisons),
        "accounts_compared": len(account_ids),
        "skipped_insufficient_data": [k for k in accounts if k not in analyzable],
        "comparisons": comparisons,
        "highest_similarity": comparisons[0] if comparisons else None,
    }

    os.makedirs(os.path.dirname(output_path) if os.path.dirname(output_path) else ".", exist_ok=True)
    with open(output_path, "w") as f:
        json.dump(output, f, indent=2, default=str)
    print(json.dumps({"comparisons": len(comparisons), "highest": comparisons[0]["cosine_similarity"] if comparisons else None}))


if __name__ == "__main__":
    if len(sys.argv) < 3:
        print(json.dumps({"error": "Usage: python3 script.py <input.json> <output.json>"}))
        sys.exit(1)
    main(sys.argv[1], sys.argv[2])
```

## Important Notes

### What This Agent Can Do
- Analyze TEXT content only. Words, sentences, punctuation, emoji.
- Compare writing styles across accounts using mathematical cosine similarity.
- Detect bots through entropy analysis and repetition detection.

### What This Agent CANNOT Do
- Analyze images, videos, audio, or visual content. If asked about visual content, respond: `VISUAL_CONTENT_NOT_SUPPORTED — this agent processes text only.`
- Determine identity with certainty. Stylometric analysis is probabilistic, not definitive.
- Analyze code writing style (that requires a different feature set).

### When to Report INSUFFICIENT_DATA
- Account has fewer than 100 total words of text
- Account has fewer than 5 posts (for syntactic analysis)
- Only bio/profile text is available (typically < 50 words)

### Ethical Note

Stylometric analysis can reveal identity across pseudonymous accounts. This capability is provided for authorized investigations only. Results should be treated as indicative, not definitive, and always corroborated with other evidence sources before making identity assertions. Always report confidence levels honestly — do not inflate them.
