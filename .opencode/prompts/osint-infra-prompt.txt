```bash
import json, sys, re

input_file = "<session_dir>/infra/crtsh_raw_<domain>.json"
output_file = "<session_dir>/infra/crtsh_parsed_<domain>.json"

try:
    with open(input_file, "r") as f:
        data = json.load(f)
except Exception as e:
    print(f"ERROR parsing crt.sh response: {e}", file=sys.stderr)
    with open(output_file, "w") as f:
        json.dump({"error": str(e), "domains": []}, f, indent=2)
    sys.exit(0)

all_domains = set()
issuers = {}
cert_details = []

for cert in data:
    # Track issuers
    issuer = cert.get("issuer_name", "unknown")
    issuers[issuer] = issuers.get(issuer, 0) + 1
    
    # Extract name_value (may contain multiple FQDNs separated by newlines)
    name_value = cert.get("name_value", "")
    for name in name_value.split("\n"):
        name = name.strip().lower()
        if name and name != "*":
            all_domains.add(name)
            
            # Handle wildcard domains — expand them
            if name.startswith("*."):
                # Log the wildcard for permutation expansion
                base = name[2:]
                all_domains.add(base)
    
    cert_details.append({
        "common_name": cert.get("common_name", ""),
        "issuer": cert.get("issuer_name", ""),
        "not_before": cert.get("not_before", ""),
        "not_after": cert.get("not_after", ""),
        "name_value": cert.get("name_value", "")
    })

# Remove duplicates and sort
unique_domains = sorted(all_domains)

result = {
    "total_certificates": len(data),
    "unique_domains": len(unique_domains),
    "domains": unique_domains,
    "issuers": issuers,
    "certificates": cert_details
}

with open(output_file, "w") as f:
    json.dump(result, f, indent=2)

print(f"crt.sh: {len(data)} certificates, {len(unique_domains)} unique domains")
for d in unique_domains:
    print(f"  CT_DOMAIN: {d}")
PYEOF
```

**Wildcard Expansion Rules**:
- For every wildcard cert like `*.example.com`, note the base domain
- These wildcards confirm the domain accepts ANY subdomain
- Cross-reference with permutation results — wildcard-cert domains should resolve if active

**Certificate Lifecycle Analysis**:

- Identify cert re-issuance patterns (same CN re-issued frequently = infrastructure churn)

**CNAME Chain Analysis** (MANDATORY for every subdomain with a CNAME):
```bash
# Follow CNAME chains recursively
cname_chain() {
    local domain="$1"
    local depth=0
    local max_depth=10
    echo -n "$domain"
    
    while [ $depth -lt $max_depth ]; do
        local target=$(dig +short CNAME "$domain" 2>/dev/null | head -1)
        if [ -z "$target" ]; then
            # No more CNAMEs — try to get A record
            local ip=$(dig +short A "$domain" 2>/dev/null | head -1)
            if [ -n "$ip" ]; then
                echo " -> A: $ip"
            else
                echo " -> NO_RESOLVE"
            fi
            break
        fi
        echo -n " -> CNAME: $target"
        domain="$target"
        depth=$((depth + 1))
    done
    
    if [ $depth -ge $max_depth ]; then
        echo " -> CNAME_CHAIN_TOO_DEEP"
    fi
}
```

**CRITICAL**: EVERY discovered subdomain MUST be resolved to an IP. No exceptions.

```bash
# Merge all sources and resolve every subdomain
python3 << 'PYEOF'
import subprocess, json, socket, sys, os

domain = "<domain>"
session_dir = "<session_dir>/infra"
all_subdomains = set()

# Collect from crt.sh
try:
    with open(f"{session_dir}/crtsh_parsed_{domain}.json") as f:

    with open(f"{session_dir}/amass_{domain}.json") as f:
        amass_data = json.load(f)
    for d in amass_data:
        if isinstance(d, dict):
            all_subdomains.add(d.get("name", ""))
        else:
            all_subdomains.add(str(d))
except: pass

# Collect from sublist3r
try:
    with open(f"{session_dir}/sublist3r_{domain}.txt") as f:
        for line in f:
            all_subdomains.add(line.strip())
except: pass

# Collect from permutations
try:
    with open(f"{session_dir}/permutations_{domain}.txt") as f:
        for line in f:
            all_subdomains.add(line.strip())
except: pass

# Remove empty entries and deduplicate
all_subdomains = {s for s in all_subdomains if s and "." in s and domain in s}

print(f"Total unique subdomains to resolve: {len(all_subdomains)}")

# Resolve every subdomain
dns_results = []
resolved_count = 0
for sub in sorted(all_subdomains):
    try:
        ips = socket.getaddrinfo(sub, None, socket.AF_INET)
        ip_set = list(set(addr[4][0] for addr in ips))
        if ip_set:
            dns_results.append({
                "subdomain": sub,
                "ips": ip_set,
                "status": "resolved"
            })
            resolved_count += 1
        else:
            dns_results.append({
                "subdomain": sub,
                "ips": [],
                "status": "no_ip"
            })
    except socket.gaierror:
        dns_results.append({
            "subdomain": sub,
            "ips": [],
            "status": "nx_domain"
        })
    except Exception as e:
        dns_results.append({
            "subdomain": sub,
            "ips": [],
            "status": f"error: {e}"
        })

print(f"Resolved: {resolved_count}/{len(all_subdomains)} subdomains")

with open(f"{session_dir}/all_subdomains_resolved_{domain}.json", "w") as f:
    json.dump(dns_results, f, indent=2)

# Output resolved subdomains for next steps
for r in dns_results:
    if r["ips"]:
        print(f"  RESOLVED: {r['subdomain']} -> {', '.join(r['ips'])}")
PYEOF
```

#### 4d. Security Header Analysis (MANDATORY)
```bash
# Security header analysis for the primary domain and all live subdomains
SEC_HEADERS_FILE="<session_dir>/infra/security_headers_<domain>.json"

python3 << 'PYEOF'
import subprocess, json, re

domain = "<domain>"
session_dir = "<session_dir>/infra"
live_file = f"{session_dir}/live_subdomains_{domain}.txt"

# Security headers to check
SECURITY_HEADERS = [
    "strict-transport-security",
    "content-security-policy",
    "x-content-type-options",
    "x-frame-options",
    "x-xss-protection",
    "referrer-policy",
    "permissions-policy",
    "cross-origin-opener-policy",
    "cross-origin-resource-policy",
    "cross-origin-embedder-policy",
    "content-security-policy-report-only",
    "x-permitted-cross-domain-policies",
    "expect-ct",
    "report-to",
    "nel"
]

results = []

def check_headers(url):
    try:
        resp = subprocess.run(
            ["curl", "-sI", "-L", "--max-time", "10", url],
            capture_output=True, text=True, timeout=15
        )
        headers = {}
        for line in resp.stdout.split("\n"):
            if ":" in line:
                key, val = line.split(":", 1)
                headers[key.strip().lower()] = val.strip()
        
        http_code = "unknown"
        try:
            code_resp = subprocess.run(
                ["curl", "-s", "-o", "/dev/null", "-w", "%{http_code}", "--max-time", "10", url],
                capture_output=True, text=True, timeout=15
            )
            http_code = code_resp.stdout.strip()
        except: pass
        
        return http_code, headers
    except Exception as e:
        return "error", {}

def analyze_security_headers(headers):
    found = {}
    missing = []
    for h in SECURITY_HEADERS:
        if h in headers:
            found[h] = headers[h]
        else:
            missing.append(h)
    return found, missing

# Check primary domain
code, hdrs = check_headers(f"https://{domain}")
found, missing = analyze_security_headers(hdrs)
results.append({
    "domain": domain,
    "http_code": code,
    "security_headers_found": found,
    "security_headers_missing": missing,
    "security_score": len(found) / len(SECURITY_HEADERS) * 100,
    "server": hdrs.get("server", "unknown"),
    "x_powered_by": hdrs.get("x-powered-by", "not disclosed")
})

# Check all live subdomains
try:
    with open(live_file) as f:
        for line in f:
            sub = line.strip()
            if not sub: continue
            code, hdrs = check_headers(f"https://{sub}")
            found, missing = analyze_security_headers(hdrs)
            results.append({
                "domain": sub,
                "http_code": code,
                "security_headers_found": found,
                "security_headers_missing": missing,
                "security_score": len(found) / len(SECURITY_HEADERS) * 100,
                "server": hdrs.get("server", "unknown"),
                "x_powered_by": hdrs.get("x-powered-by", "not disclosed")
            })
except Exception as e:
    print(f"Error reading live subdomains: {e}")

with open(f"{session_dir}/security_headers_{domain}.json", "w") as f:
    json.dump(results, f, indent=2)

print(f"Security header analysis: {len(results)} domains checked")
for r in results:
    print(f"  {r['domain']}: score={r['security_score']:.0f}% missing={len(r['security_headers_missing'])} headers")
PYEOF
```

- Paste sites: Minimum 2 seconds between requests

- Do NOT attempt directory brute-forcing beyond the 50+ defined paths
- Do NOT attempt SQL injection or vulnerability scanning
- If you discover an exposed .git directory, note it but do NOT clone the repository
